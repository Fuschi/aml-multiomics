prepare_data

Reads raw files from all three cohorts, aligns samples to WES IDs, and writes clean matrices to processed/.

library(tidyverse)
library(knitr)
library(readxl)

WES mutations

Long-format somatic mutation table shared across BEAT-AML, TCGA, and NGS-PTL. TCGA sample IDs have their -03 suffix stripped to match the format used in the expression data.

wes <- read_xlsx("../raw/WES_filtered_final.xlsx") |>
  mutate(ID = if_else(cohort == "TCGA", str_remove(ID, "-03"), ID))

# Binary mutation matrix
wes_wide <- wes %>%
  distinct(ID, gene_symbol) %>%
  mutate(mutated = TRUE) %>%
  pivot_wider(names_from  = gene_symbol,
              values_from = mutated,
              values_fill = FALSE) %>%
  rename(sample_id = ID)

BEAT-AML expression

The expression matrix uses RNA-seq IDs (dbgap_rnaseq_sample). The sample mapping table (source: Bottomly et al. 2022) converts these to WES IDs (labId), keeping only samples flagged for inclusion.

expr_beat <- read_tsv("../raw/210126_BEAT-AML_CPM-TMM_log2_with_hugo.tsv.xz",
                      show_col_types = FALSE) |>
  column_to_rownames("symbol") |>
  t() |>
  as_tibble(rownames = "dbgap_rnaseq_sample")

meta_beat <- read_xlsx("../raw/beataml_waves1to4_sample_mapping.xlsx") |>
  filter(rna_include_in_analysis == "yes")

expr_beat <- expr_beat |>
  left_join(select(meta_beat, dbgap_rnaseq_sample, labId),
            by = "dbgap_rnaseq_sample") |>
  relocate(labId, .after = "dbgap_rnaseq_sample") |>
  select(-dbgap_rnaseq_sample)

Gene overlap: BEAT-AML vs TCGA-AML

The two cohorts were normalised independently and cannot be merged. Genes exclusive to BEAT-AML are nearly all zero (>99% zero counts) and will be dropped when filtering to Human-GEM metabolic genes.

ggVennDiagram(x = list(beat = colnames(expr_beat_wes)[-1],
                       tcga = colnames(expr_tcga_wes)[-1]))
local({
  
  inter <- intersect(colnames(expr_beat_wes)[-1], colnames(expr_tcga_wes)[-1])
  
  only_beat <- setdiff(colnames(expr_beat_wes)[-1], colnames(expr_tcga_wes)[-1])
  only_tcga <- setdiff(colnames(expr_tcga_wes)[-1], colnames(expr_beat_wes)[-1])
  
  tibble(
    cohort = c("BEAT-AML", "TCGA-LAML"),
    
    wes_samples = c(
      wes |> filter(cohort == "BEAT-AML") |> pull(ID) |> unique() |> length(),
      wes |> filter(cohort == "TCGA") |> pull(ID) |> unique() |> length()
    ),
    
    rna_samples = c(
      n_distinct(expr_beat$labId),
      n_distinct(expr_tcga$sample_id)
    ),
    
    matched_rna_wes_samples = c(
      n_distinct(expr_beat_wes$labId),
      n_distinct(expr_tcga_wes$sample_id)
    ),
    
    total_genes = c(
      ncol(expr_beat_wes) - 1,
      ncol(expr_tcga_wes) - 1
    ),
    
    shared_genes = length(inter),
    
    cohort_specific_genes = c(
      length(only_beat),
      length(only_tcga)
    ),
    
    zero_fraction_in_shared_genes = c(
      mean(as.matrix(expr_beat_wes[, inter, drop = FALSE]) == 0, na.rm = TRUE),
      mean(as.matrix(expr_tcga_wes[, inter, drop = FALSE]) == 0, na.rm = TRUE)
    ),
    
    shared_expression_fraction = c(
      sum(as.matrix(expr_beat_wes[, inter, drop = FALSE]), na.rm = TRUE) /
        sum(as.matrix(expr_beat_wes[, -1, drop = FALSE]), na.rm = TRUE),
      
      sum(as.matrix(expr_tcga_wes[, inter, drop = FALSE]), na.rm = TRUE) /
        sum(as.matrix(expr_tcga_wes[, -1, drop = FALSE]), na.rm = TRUE)
    ),
    
    zero_fraction_in_cohort_specific_genes = c(
      if (length(only_beat) == 0) {
        NA_real_
      } else {
        mean(as.matrix(expr_beat_wes[, only_beat, drop = FALSE]) == 0, na.rm = TRUE)
      },
      
      if (length(only_tcga) == 0) {
        NA_real_
      } else {
        mean(as.matrix(expr_tcga_wes[, only_tcga, drop = FALSE]) == 0, na.rm = TRUE)
      }
    )
  ) |>
    mutate(
      zero_fraction_in_shared_genes = scales::percent(
        zero_fraction_in_shared_genes,
        accuracy = 0.01
      ),
      shared_expression_fraction = scales::percent(
        shared_expression_fraction,
        accuracy = 0.01
      ),
      zero_fraction_in_cohort_specific_genes = scales::percent(
        zero_fraction_in_cohort_specific_genes,
        accuracy = 0.01
      )
    ) |>
    knitr::kable(
      col.names = c(
        "Cohort",
        "WES samples",
        "RNA samples",
        "Matched RNA-WES samples",
        "Total genes",
        "Shared genes",
        "Cohort-specific genes",
        "Zero values in shared genes (%)",
        "Total expression found in shared genes (%)",
        "Zero values in cohort-specific genes (%)"
      ),
      align = c("l", rep("c", 9))
    )
})

Metadata

Sample-level metadata assembled from the karyotype file and the BEAT-AML mapping table. Rows are sorted to match the expression matrix row order.

karyotype <- read_xlsx("../raw/Karyotype and other variables.xlsx", col_types = "text") |>
  select(-2, -6) |>
  rename(Disease_Type = Disease.type, sample_id = ID)
beat_wes_metadata <- meta_beat |>
  filter(labId %in% expr_beat_wes$labId) |>
  relocate(labId) |>
  rename(sample_id = labId) |>
  left_join(
    karyotype |>
      filter(cohort == "BEAT-AML") |>
      select(sample_id, Patient_Id, Gender, Disease_Type),
    by = "sample_id"
  ) |>
  arrange(match(sample_id, expr_beat_wes$labId)) %>%
  mutate(cohort = "BEAT")

tcga_wes_metadata <- karyotype |>
  filter(cohort == "TCGA") |>
  filter(sample_id %in% expr_tcga_wes$sample_id) |>
  arrange(match(sample_id, expr_tcga_wes$sample_id))

write_tsv(beat_wes_metadata,  "../processed/beat_wes_metadata.tsv")
write_tsv(tcga_wes_metadata, "../processed/tcga_wes_metadata.tsv")
