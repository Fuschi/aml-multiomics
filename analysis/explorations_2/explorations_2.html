Cross-cohort comparison of BEAT-AML and TCGA-LAML

1 Introduction

This notebook presents a step-by-step exploratory analysis of metabolic gene expression in acute myeloid leukemia (AML), comparing two independent cohorts: BEAT-AML and TCGA-LAML.

The analysis focuses on genes annotated in the Human-GEM metabolic model. The main aim is to characterize metabolic expression patterns, evaluate the overlap between cohorts, and prepare the data for downstream multivariate and pathway-level analyses.

2 Setup

3 Data loading

We load pre-processed expression matrices for BEAT-AML and TCGA-LAML, restricted to samples with matched whole-exome sequencing information.

The BEAT-AML expression matrix contains log2 TMM-CPM normalized values, while the TCGA-LAML expression matrix contains linear TMM-CPM normalized values. Both datasets use HUGO gene symbols.

Gene-level metabolic annotation from Human-GEM is used to restrict downstream analyses to genes linked to known metabolic reactions and subsystems.

4 Dataset overview

Before focusing on metabolic genes, we summarize the overlap between the BEAT-AML and TCGA-LAML expression matrices. We report the number of matched RNA-WES samples, shared and cohort-specific genes, and the fraction of expression signal and zero values associated with each gene set.

local({
  
  inter <- intersect(colnames(expr_beat_wes)[-1], colnames(expr_tcga_wes)[-1])

  only_tcga <- setdiff(colnames(expr_tcga_wes)[-1], colnames(expr_beat_wes)[-1])

    cohort = c("BEAT-AML", "TCGA-LAML"),
    
    matched_rna_wes_samples = c(
      n_distinct(expr_beat_wes$sample_id),
      n_distinct(expr_tcga_wes$sample_id)
    ),
    
    total_genes = c(
      ncol(expr_beat_wes) - 1,
      ncol(expr_tcga_wes) - 1
    ),
    
    shared_genes = length(inter),
    
    cohort_specific_genes = c(
      length(only_beat),
      length(only_tcga)
    ),
    
    zero_fraction_in_shared_genes = c(
      mean(as.matrix(expr_beat_wes[, inter, drop = FALSE]) == 0, na.rm = TRUE),
      mean(as.matrix(expr_tcga_wes[, inter, drop = FALSE]) == 0, na.rm = TRUE)
    ),
    
    shared_expression_fraction = c(
      sum(as.matrix(expr_beat_wes[, inter, drop = FALSE]), na.rm = TRUE) /

5 Metabolic gene overview

6 Create long-format expression table

As a starting point, we retain only genes shared between BEAT-AML and TCGA-LAML. These shared genes account for more than 99% of the total expression signal in both cohorts and provide a common feature for downstream analyses.

Show code 
long_shared_aml <- local({
  
  shared_all_genes <- intersect(colnames(expr_beat_wes)[-1], colnames(expr_tcga_wes)[-1])
  
  bind_rows(
    expr_beat_wes |>
      select(sample_id, all_of(shared_all_genes)) |>
      pivot_longer(-sample_id, names_to = "hugo", values_to = "log2_tmm_cpm") |>
      mutate(tmm_cpm = 2^log2_tmm_cpm - 1, .before = "log2_tmm_cpm") |>
      left_join(gem, by = "hugo") |>
      left_join(select(meta_beat_wes, cohort, sample_id, Patient_Id, Gender, Disease_Type), by = "sample_id"),
    
    expr_tcga_wes |>
      select(sample_id, all_of(shared_all_genes)) |>
      pivot_longer(-sample_id, names_to = "hugo", values_to = "tmm_cpm") |>
      mutate(log2_tmm_cpm = log2(tmm_cpm + 1)) |>
      left_join(gem, by = "hugo") |>
      left_join(meta_tcga_wes, by = "sample_id")
  ) |>
    mutate(
      hugo_fct = if_else(hugo %in% gem$hugo, hugo, "non_metabolic"),
      subsystems = if_else(hugo_fct == "non_metabolic", "non_metabolic", subsystems),
      pathway_class = if_else(hugo_fct == "non_metabolic", "non_metabolic", pathway_class)
    ) |>
    relocate(hugo_fct, .after = "hugo")
})

7 Select main pathway classes

To simplify pathway-level visualization, we retain the most represented pathway classes and group the remaining metabolic pathways as other_metabolic.

Show code 
top_pathway_class <- long_shared_aml |>
  filter(pathway_class != "non_metabolic") |>
  group_by(sample_id, pathway_class) |>
  summarise(abun_metab = sum(log2_tmm_cpm), .groups = "drop") |>
  group_by(sample_id) |>
  mutate(rela_metab = abun_metab / sum(abun_metab)) |>
  ungroup() |>
  summarise(total_rela = sum(rela_metab), .by = pathway_class) |>
  slice_max(total_rela, n = 20) |>
  pull(pathway_class)

long_shared_aml <- long_shared_aml |>
  mutate(
    pathway_class_fct = case_when(
      is.na(pathway_class) ~ NA_character_,
      pathway_class == "non_metabolic" ~ "non_metabolic",
      pathway_class %in% top_pathway_class ~ pathway_class,
      TRUE ~ "other_metabolic"

      pathway_class_fct,

Show plot function 
plot_stacked_bar <- function(data,
                             include_non_metabolic = TRUE,
                             relative              = FALSE,
                             title                 = NULL) {

  if (!include_non_metabolic) {
    data <- data |> filter(pathway_class_fct != "non_metabolic")
  }

  data <- data |>
    group_by(cohort, sample_id, pathway_class_fct) |>
    summarise(log2_tmm_cpm = sum(log2_tmm_cpm), .groups = "drop")

  if (relative) {
    data <- data |>
      group_by(cohort, sample_id) |>
      mutate(y_var = log2_tmm_cpm / sum(log2_tmm_cpm)) |>
      ungroup()
    y_label <- "Relative pathway signal"
  } else {
    data <- data |> mutate(y_var = log2_tmm_cpm)
    y_label <- "log2(TMM-CPM+1)"
  }

  data |>
    ggplot(aes(x = sample_id, y = y_var, fill = pathway_class_fct)) +
    geom_col() +
    theme_minimal() +
    theme(
      axis.text.x     = element_blank(),
      axis.ticks.x    = element_blank(),
      legend.position = "top"
    ) +
    labs(
      x     = "Samples",
      y     = y_label,
      fill  = "Pathway class",
      title = title
    ) +
    facet_wrap(~cohort, nrow = 2, scales = "free") +
    ggqualpal::scale_fill_qualpal()
}

9 Metabolic matrix construction and batch correction

From this point onward, we restrict the analysis to metabolic genes only and build the matrices used for dimensionality reduction and following analysis in wide format.

long_aml_metab <- long_shared_aml |>
  filter(hugo_fct != "non_metabolic")

wide_metab <- local({
  
  wide_all <- long_aml_metab |>
    group_by(hugo) |>
    filter(n_distinct(cohort) == 2) |>
    ungroup() |>
    select(sample_id, cohort, Disease_Type, Gender, hugo_fct, log2_tmm_cpm) |>
    pivot_wider(names_from = hugo_fct, values_from = log2_tmm_cpm)
  
  meta_beat <- meta_beat_wes |>
    filter(sample_id %in% wide_all$sample_id) |>
    arrange(sample_id)
  
  meta_tcga <- meta_tcga_wes |>
    filter(sample_id %in% wide_all$sample_id) |>
    arrange(sample_id)
  
  wide_beat <- wide_all |>
    filter(cohort == "BEAT") |>
    arrange(sample_id)
  
  wide_tcga <- wide_all |>
    filter(cohort == "TCGA") |>
    arrange(sample_id)
  
  wide_all <- bind_rows(wide_beat, wide_tcga)
  
  meta_aml <- wide_all |>
    select(sample_id, cohort, Disease_Type, Gender)
  
  metab_genes <- setdiff(colnames(wide_all), colnames(meta_aml))
  
  mat_aml <- wide_all |>
    select(sample_id, all_of(metab_genes)) |>
    column_to_rownames("sample_id") |>
    as.matrix()
  
  mat_beat <- wide_beat |>
    select(sample_id, all_of(metab_genes)) |>
    column_to_rownames("sample_id") |>
    as.matrix()
  
  mat_tcga <- wide_tcga |>
    select(sample_id, all_of(metab_genes)) |>
    column_to_rownames("sample_id") |>
    as.matrix()
  
  stopifnot(identical(rownames(mat_beat), meta_beat$sample_id))
  stopifnot(identical(rownames(mat_tcga), meta_tcga$sample_id))
  stopifnot(identical(rownames(mat_aml), meta_aml$sample_id))
  
  list(
    wide_all = wide_all,
    meta_aml = meta_aml,
    meta_beat = meta_beat,
    meta_tcga = meta_tcga,
    mat_aml = mat_aml,
    mat_beat = mat_beat,
    mat_tcga = mat_tcga
  )
})

pca_plot <- function(pca_obj, color_vec, title, palette = NULL) {
  
  n_levels <- length(unique(color_vec))
  
  if (is.null(palette)) {
    palette <- ggpubr::get_palette("jco", n_levels)
  }
  
  factoextra::fviz_pca_biplot(
    pca_obj,
    geom.ind     = "point",
    col.ind      = color_vec,
    addEllipses  = n_levels > 1,
    ellipse.type = "norm",
    geom.var     = c("arrow", "text"),
    col.var      = "darkgray",
    select.var   = list(cos2 = 20),
    repel        = TRUE,
    palette      = palette,
    legend.title = "",
    title        = title
  ) +
    theme_bw() +
    coord_equal()
}

# Introduction

This notebook presents a step-by-step exploratory analysis of metabolic gene expression in acute myeloid leukemia (AML), comparing two independent cohorts: BEAT-AML and TCGA-LAML.

The analysis focuses on genes annotated in the Human-GEM metabolic model. The main aim is to characterize metabolic expression patterns, evaluate the overlap between cohorts, and prepare the data for downstream multivariate and pathway-level analyses.

# Data loading

We load pre-processed expression matrices for BEAT-AML and TCGA-LAML, restricted to samples with matched whole-exome sequencing information.

The BEAT-AML expression matrix contains log2 TMM-CPM normalized values, while the TCGA-LAML expression matrix contains linear TMM-CPM normalized values. Both datasets use HUGO gene symbols.

Gene-level metabolic annotation from Human-GEM is used to restrict downstream analyses to genes linked to known metabolic reactions and subsystems.

# Dataset overview

Before focusing on metabolic genes, we summarize the overlap between the BEAT-AML and TCGA-LAML expression matrices. We report the number of matched RNA-WES samples, shared and cohort-specific genes, and the fraction of expression signal and zero values associated with each gene set.

```{r}
#| label: gene-and-sample-overlap-summary

local({
  
  inter <- intersect(colnames(expr_beat_wes)[-1], colnames(expr_tcga_wes)[-1])

  only_tcga <- setdiff(colnames(expr_tcga_wes)[-1], colnames(expr_beat_wes)[-1])
  
  tibble(
    cohort = c("BEAT-AML", "TCGA-LAML"),

      n_distinct(expr_beat_wes$sample_id),

      ncol(expr_beat_wes) - 1,
      ncol(expr_tcga_wes) - 1

    cohort_specific_genes = c(
      length(only_beat),
      length(only_tcga)

    zero_fraction_in_shared_genes = c(
      mean(as.matrix(expr_beat_wes[, inter, drop = FALSE]) == 0, na.rm = TRUE),
      mean(as.matrix(expr_tcga_wes[, inter, drop = FALSE]) == 0, na.rm = TRUE)
    ),
    
    shared_expression_fraction = c(
      sum(as.matrix(expr_beat_wes[, inter, drop = FALSE]), na.rm = TRUE) /
        sum(as.matrix(expr_beat_wes[, -1, drop = FALSE]), na.rm = TRUE),
      
      sum(as.matrix(expr_tcga_wes[, inter, drop = FALSE]), na.rm = TRUE) /
        sum(as.matrix(expr_tcga_wes[, -1, drop = FALSE]), na.rm = TRUE)

      } else {
```

```{r}
plot_stacked_bar <- function(data,
                             include_non_metabolic = TRUE,
                             relative              = FALSE,
                             title                 = NULL) {

  if (!include_non_metabolic) {
    data <- data |> filter(pathway_class_fct != "non_metabolic")
  }

  data <- data |>
    group_by(cohort, sample_id, pathway_class_fct) |>
    summarise(log2_tmm_cpm = sum(log2_tmm_cpm), .groups = "drop")

  if (relative) {
    data <- data |>
      group_by(cohort, sample_id) |>
      mutate(y_var = log2_tmm_cpm / sum(log2_tmm_cpm)) |>
      ungroup()
    y_label <- "Relative pathway signal"
  } else {
    data <- data |> mutate(y_var = log2_tmm_cpm)
    y_label <- "log2(TMM-CPM+1)"
  }

  data |>
    ggplot(aes(x = sample_id, y = y_var, fill = pathway_class_fct)) +
    geom_col() +
    theme_minimal() +

      axis.text.x     = element_blank(),
      axis.ticks.x    = element_blank(),

    ) +

      x     = "Samples",
      y     = y_label,
      fill  = "Pathway class",
      title = title
    ) +
    facet_wrap(~cohort, nrow = 2, scales = "free") +
```

## UMAP

```{r}
library(uwot)

umap_plot <- function(umap_mat, color_vec, title, palette = NULL) {
  
  n_levels <- length(unique(color_vec))
  
  if (is.null(palette)) {
    palette <- ggpubr::get_palette("jco", n_levels)
  }
  
  p <- tibble(
    UMAP1 = umap_mat[, 1],
    UMAP2 = umap_mat[, 2],
    group = color_vec
  ) %>%
    ggplot(aes(UMAP1, UMAP2, color = group)) +
    geom_point(alpha = 0.7, size = 2) +
    scale_color_manual(values = palette) +
    theme_bw() +
    coord_equal() +
    labs(title = title, color = "")
  
  if (n_levels > 1) {
    p <- p + stat_ellipse(type = "norm", linewidth = 0.8)
  }
  
  p
}
```
